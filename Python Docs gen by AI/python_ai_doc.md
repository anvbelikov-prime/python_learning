# 1. Синтаксис и базовые конструкции

## Теоретическая часть

Python — интерпретируемый язык программирования с динамической типизацией. Все данные в Python представлены объектами, каждый из которых имеет тип, определяющий доступные операции.

### Ключевые понятия и определения

| Понятие | Определение |
|---------|-------------|
| **Литерал** | Фиксированное значение в коде: 42, "Hello", True |
| **Выражение** | Комбинация значений, переменных и операторов, вычисляющаяся в результат |
| **Оператор** | Символ или ключевое слово, выполняющие операцию (например, +, -, *) |
| **int (целое число)** | Тип данных для целых чисел: 42, -17, 0 |
| **float** | Тип для чисел с плавающей точкой: 3.14, -0.5 |
| **bool** | Логический тип: True или False |
| **str (строка)** | Неизменяемая последовательность символов в кавычках |
| **if/elif/else** | Условные операторы для ветвления программы |
| **while** | Цикл, повторяющий блок кода пока условие истинно |
| **for** | Цикл для итерации по элементам последовательности |
| **in** | Оператор для проверки вхождения элемента |

## Примеры кода

```python
# Базовые типы данных и литералы
целое_число = 42  # int
дробное_число = 3.14  # float
логическое_значение = True  # bool
строка = "Привет, Python!"  # str

# Проверка типа
print(type(целое_число))  # <class 'int'>
print(isinstance(дробное_число, float))  # True

# Арифметические операции
сумма = 10 + 5  # 15
разность = 10 - 5  # 5
произведение = 10 * 5  # 50
частное = 10 / 3  # 3.3333...
целочисленное_деление = 10 // 3  # 3
остаток = 10 % 3  # 1
степень = 2 ** 3  # 8

# Операторы сравнения
равно = (5 == 5)  # True
не_равно = (5 != 3)  # True
больше = (10 > 5)  # True
меньше_или_равно = (5 <= 10)  # True

# Логические операторы
результат_and = True and False  # False
результат_or = True or False  # True
результат_not = not True  # False

# Условный оператор
возраст = 25
if возраст >= 18:
    print("Совершеннолетний")
elif возраст >= 13:
    print("Подросток")
else:
    print("Ребёнок")

# Цикл while
счетчик = 0
while счетчик < 5:
    print(f"Счётчик: {счетчик}")
    счетчик += 1

# Цикл for
числа = [1, 2, 3, 4, 5]
for число in числа:
    print(f"Квадрат {число} = {число ** 2}")

# Оператор in
if "Python" in "Привет, Python!":
    print("Подстрока найдена")

# Функция range() для генерации последовательностей
for i in range(3):  # 0, 1, 2
    print(i)

for i in range(1, 6):  # 1, 2, 3, 4, 5
    print(i)

for i in range(0, 10, 2):  # 0, 2, 4, 6, 8 (с шагом 2)
    print(i)
```

# 2. Коллекции и работа с данными в памяти

## Теоретическая часть

Python предоставляет встроенные структуры данных (коллекции) для работы с наборами элементов. Каждая коллекция имеет свои характеристики: изменяемость, упорядоченность, уникальность элементов.

### Ключевые понятия и определения

| Понятие | Определение | Пример |
|---------|-------------|---------|
| **list (список)** | Упорядоченная изменяемая коллекция | [1, 2, 3] |
| **tuple (кортеж)** | Упорядоченная неизменяемая коллекция | (1, 2, 3) |
| **dict (словарь)** | Неупорядоченная коллекция пар ключ-значение | {"имя": "Анна"} |
| **set (множество)** | Неупорядоченная коллекция уникальных элементов | {1, 2, 3} |
| **Изменяемость** | Способность менять содержимое после создания | list, dict, set изменяемы |
| **Срезы** | Получение подпоследовательности: [start:stop:step] | [1:4], [::2] |
| **Распаковка** | Присваивание элементов коллекции переменным | x, y, z = (1, 2, 3) |

## Примеры кода

```python
# ============ СПИСКИ (list) ============
список = [1, 2, 3, 4, 5]
список.append(6)  # Добавление элемента
список.insert(0, 0)  # Вставка на позицию
элемент = список.pop()  # Удаление последнего элемента
список.remove(3)  # Удаление элемента по значению
список.sort()  # Сортировка на месте
список.reverse()  # Переворот списка

# Доступ по индексу и срезы
первый = список[0]  # Первый элемент
последний = список[-1]  # Последний элемент
подсписок = список[1:4]  # Срез [1, 2, 3]
каждый_второй = список[::2]  # Каждый второй элемент
обратный = список[::-1]  # В обратном порядке

# ============ КОРТЕЖИ (tuple) ============
кортеж = (1, 2, 3, 4, 5)  # Неизменяемый
одноэлементный = (42,)  # Обратите внимание на запятую!

# Распаковка
x, y, z = (1, 2, 3)
первый, *средние, последний = (1, 2, 3, 4, 5)
# первый = 1, средние = [2, 3, 4], последний = 5

# ============ СЛОВАРИ (dict) ============
словарь = {"имя": "Иван", "возраст": 30, "город": "Москва"}
словарь["профессия"] = "аналитик"  # Добавление

# Безопасное получение значения (возвращает None если ключа нет)
возраст = словарь.get("возраст", 0)

# Удаление
удалённое_значение = словарь.pop("город")

# Итерация
for ключ in словарь:  # По ключам
    print(ключ, словарь[ключ])

for ключ, значение in словарь.items():  # По парам
    print(f"{ключ}: {значение}")

# ============ МНОЖЕСТВА (set) ============
множество1 = {1, 2, 3, 4, 5}
множество2 = {4, 5, 6, 7, 8}

объединение = множество1 | множество2  # {1, 2, 3, 4, 5, 6, 7, 8}
пересечение = множество1 & множество2  # {4, 5}
разность = множество1 - множество2  # {1, 2, 3}
симметричная = множество1 ^ множество2  # {1, 2, 3, 6, 7, 8}

множество1.add(6)  # Добавление элемента
множество1.remove(1)  # Удаление (ошибка если нет)
множество1.discard(10)  # Удаление (без ошибки если нет)

# ============ МЕТОДЫ КОЛЛЕКЦИЙ ============
# Методы списков
числа = [3, 1, 4, 1, 5, 9, 2, 6]
print(числа.count(1))  # Количество вхождений: 2
print(числа.index(4))  # Индекс первого вхождения: 2
числа.extend([5, 3, 5])  # Добавление элементов из другого списка

# Методы строк (строки тоже последовательности)
текст = "  Привет, Python!  "
print(текст.upper())  # "  ПРИВЕТ, PYTHON!  "
print(текст.lower())  # "  привет, python!  "
print(текст.strip())  # "Привет, Python!" (без пробелов по краям)
```

```python
def найти_уникальные(список):
    видели = set()
    результат = []
    for элемент in список:
        if элемент not in видели:
            видели.add(элемент)
            результат.append(элемент)
    return результат

# Тест
```

# 3. Функции и организация кода

## Теоретическая часть

Функции позволяют организовать повторно используемый код. Python поддерживает различные параметры функций и мощное правило LEGB для разрешения области видимости.

### Ключевые понятия и определения

| Понятие | Определение |
|---------|-------------|
| **Функция** | Именованный блок кода, выполняющий определённую задачу |
| **def** | Ключевое слово для объявления функции |
| **Параметры** | Переменные в определении функции |
| **Аргументы** | Фактические значения при вызове функции |
| **Позиционные аргументы** | Передаются в порядке определения |
| **Именованные аргументы** | Передаются с указанием имени параметра |
| **\*args** | Произвольное количество позиционных аргументов (кортеж) |
| **\*\*kwargs** | Произвольное количество именованных аргументов (словарь) |
| **LEGB** | Порядок поиска переменных: Local → Enclosing → Global → Built-in |
| **Аннотации типов** | Указания типов параметров и возвращаемого значения |

## Примеры кода

```python
# Простая функция
def приветствие():
    """Функция выводит приветствие."""
    print("Привет, мир!")

приветствие()  # Вызов функции

# Функция с параметрами и возвращаемым значением
def сумма(a, b):
    """Возвращает сумму двух чисел."""
    return a + b

результат = сумма(5, 3)  # 8

# Функция с параметром по умолчанию
def возведение_в_степень(число, степень=2):
    """Возводит число в степень (по умолчанию в квадрат)."""
    return число ** степень

print(возведение_в_степень(3))  # 9 (степень = 2)
print(возведение_в_степень(3, 3))  # 27

# *args - произвольное количество позиционных аргументов
def сумма_всех(*числа):
    """Возвращает сумму всех переданных чисел."""
    return sum(числа)

print(сумма_всех(1, 2, 3, 4, 5))  # 15

# **kwargs - произвольное количество именованных аргументов
def информация(**данные):
    """Выводит все переданные именованные аргументы."""
    for ключ, значение in данные.items():
        print(f"{ключ}: {значение}")

информация(имя="Мария", возраст=28, профессия="аналитик")

# Комбинация различных типов параметров
def сложная_функция(обязательный, по_умолчанию="default", *args, **kwargs):
    """Демонстрация различных типов параметров."""
    print(f"Обязательный: {обязательный}")
    print(f"По умолчанию: {по_умолчанию}")
    print(f"Позиционные: {args}")
    print(f"Именованные: {kwargs}")

# Область видимости LEGB
глобальная_переменная = "глобальная"

def внешняя_функция():
    объемлющая_переменная = "объемлющая"
    
    def внутренняя_функция():
        локальная_переменная = "локальная"
        print(локальная_переменная)  # Local
        print(объемлющая_переменная)  # Enclosing
        print(глобальная_переменная)  # Global
        print(len)  # Built-in
    
    внутренняя_функция()

внешняя_функция()

# Аннотации типов (Python 3.5+)
def умножение(a: int, b: int) -> int:
    """Умножает два целых числа и возвращает результат."""
    return a * b

# Функция как объект
def применить_функцию(функция, значение):
    """Применяет переданную функцию к значению."""
    return функция(значение)

def квадрат(x):
    return x ** 2

результат = применить_функцию(квадрат, 5)  # 25
```

Напишите функцию `факториал(n)`, которая возвращает факториал числа n, используя рекурсию.

```python
def факториал(n):
    """Возвращает факториал числа n."""
    if n < 0:
        raise ValueError("Факториал не определен для отрицательных чисел")
    if n == 0 or n == 1:
        return 1
    return n * факториал(n - 1)

# Тесты
```


### Упражнение 3.3: Простой декоратор (без @ синтаксиса)

Напишите функцию-декоратор, которая принимает другую функцию и возвращает новую функцию, выводящую время выполнения.

**Решение:**

```python
import time
import functools

def отсчитать_время(функция):
    """Декоратор для измерения времени выполнения функции."""
    @functools.wraps(функция)
    def обёртка(*args, **kwargs):
        начало = time.time()
        результат = функция(*args, **kwargs)
        конец = time.time()
        затраченное_время = конец - начало
        print(f"Функция '{функция.__name__}' выполнена за {затраченное_время:.4f} сек")
        return результат
    return обёртка

# Использование
@отсчитать_время
def медленная_функция():
    time.sleep(1)
    return "Готово"

медленная_функция()
```

# 4. Строки, файлы и базовый I/O

## Теоретическая часть

Строки в Python — неизменяемые последовательности символов. Работа с файлами осуществляется через функцию `open()`, которая возвращает файловый объект. Контекстный менеджер `with` обеспечивает автоматическое закрытие файла после завершения работы.

### Ключевые понятия и определения

| Понятие | Определение |

# 7. Итераторы и генераторы

## Теоретическая часть

## Примеры кода

```python
# ПРОСТОЙ ИТЕРАТОР
class СчетчикДо:
    def __init__(self, максимум):
        self.максимум = максимум
        self.текущее = 0
    
    def __iter__(self):
        return self
    
    def __next__(self):
        если self.текущее < self.максимум:
            результат = self.текущее
            self.текущее += 1
            return результат
        else:
            raise StopIteration

# ГЕНЕРАТОРНАЯ ФУНКЦИЯ
def генератор_чисел(n):
    """Генератор чисел от 0 до n-1."""
    число = 0
    while число < n:
        yield число
        число += 1

# ГЕНЕРАТОР ФИБОНАЧЧИ
def генератор_фибоначчи(n):
    """Генератор первых n чисел Фибоначчи."""
    a, b = 0, 1
    count = 0
    while count < n:
        yield a
        a, b = b, a + b
        count += 1

# ITERTOOLS
from itertools import chain, cycle, islice

# chain - объединение
объединенный = chain([1, 2], [3, 4], [5, 6])

# cycle - бесконечное повторение
циклический = cycle([1, 2, 3])

# islice - срез
срез = islice(range(100), 10, 20, 2)
```

```python
            кеш[args] = функция(*args)
        return кеш[args]
    
    return обёртка
```

```python
import asyncio
```
